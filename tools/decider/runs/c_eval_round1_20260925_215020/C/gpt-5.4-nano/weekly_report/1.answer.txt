[1] variable.set(Name=%errors%, Value=[])
[2] loop.foreach(Collection=%files%); goal.call(Name="ScanFile", file=%item%)
[3] condition.if(Left=%errors%, Operator="isempty")

[5] goal.return()
[6] condition.if(Left=%files%, Operator="is", Right="list")
[7] output.write(Data="Scanned the log files")
[8] ui.render(Template="report.html", Parameter=[{errors: %errors%}])
[9] file.save(Path="/reports/week.html", Value=%html%)

[11] output.write(Data="Report sent", channel="log")
