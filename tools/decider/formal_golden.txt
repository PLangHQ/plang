# checkout
[0] file.read(Path="orders/%orderId%.json"); variable.set(Name=%order%, Value=%!data%)
[1] list.count(ListName=%order.items%); variable.set(Name=%itemCount%, Value=%!data%)
[2] condition.if(Left=%itemCount%, Operator="==", Right=0) { output.write(Data="Your cart is empty") }; condition.elseif(Left=%itemCount%, Operator="==", Right=1) { output.write(Data="One item") }; condition.else() { output.write(Data="%itemCount% items") }
[3] condition.if(Left=%order.email%, Operator="notcontains", Right="@") { error.throw(Message="Invalid email address") }
[4] condition.if(Left=%order.coupon%, Operator="isnotempty") { goal.call(Name="ApplyCoupon", Parameter={code: %order.coupon%, order: %order%}) }
[5] condition.if(Left=%order.total%, Operator="isnot", Right="number") { error.throw(Message="The total must be a number") }
[6] condition.if(Left=%order.country%, Operator="==", Right="IS") { variable.set(Name=%vat%, Value=0.24) }; condition.else() { variable.set(Name=%vat%, Value=0.25) }
[7] math.multiply(A=%order.total%, B=%vat%); variable.set(Name=%vatAmount%, Value=%!data%)
[8] condition.if(Left=%order.note%, Operator="notstartswith", Right="#") { list.add(ListName=%notes%, Value=%order.note%) }
[9] output.write(Data="Total: %order.total%, VAT: %vatAmount%")
